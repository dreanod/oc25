\section{Notations et définitions}

\begin{definition}[Complexité asymptotique]
  On dit que $f(n) = O(g(n))$ s'il existe $c > 0$ et $n_0 \in \mathbb{N}$
  tels que $f(n) \leq c \cdot g(n)$ pour tout $n \geq n_0$.
\end{definition}

\begin{theorem}[Borne inférieure du tri]
  Tout algorithme de tri par comparaison requiert $\Omega(n \log n)$
  comparaisons dans le pire cas.
\end{theorem}

\begin{proof}
  L'arbre de décision d'un algorithme de tri sur $n$ éléments possède
  au moins $n!$ feuilles, donc une hauteur au moins $\log_2(n!) = \Omega(n \log n)$.
\end{proof}

\section{Exemple de code}

\begin{lstlisting}[language=Python, caption={Recherche KMP}]
def kmp_search(text, pattern):
    """Knuth-Morris-Pratt string matching."""
    n, m = len(text), len(pattern)
    lps = compute_lps(pattern)  # Longest proper prefix-suffix
    i = j = 0
    while i < n:
        if pattern[j] == text[i]:
            i += 1; j += 1
        if j == m:
            return i - j  # Match at position i-j
        elif i < n and pattern[j] != text[i]:
            j = lps[j - 1] if j != 0 else 0; i += (1 if j == 0 else 0)
    return -1  # Not found
\end{lstlisting}
